                _r0 = PRQ["rec"]

                if _ps is None or int(_ps.quantity or 0) <= 0 or _ps.symbol != _r0["sym"]:
                    PRLOG.append(dict(ev="PROBE_CLOSED", t=now.isoformat(), sym=_r0["sym"]))
                    PRQ["rec"] = None
                elif now > _r0["deadline"]:
                    PRLOG.append(dict(ev="PROBE_EXPIRE", t=now.isoformat(), sym=_r0["sym"]))
                    PRQ["rec"] = None
                else:
                    _ok, _ti = _trend_ok(now, _r0, bars_3m)
                    if _ti is not None and not _ti.get("c", True):
                        PRLOG.append(dict(ev="PROBE_ABORT_BOX", t=now.isoformat(), sym=_r0["sym"], **_ti))
                        PRQ["rec"] = None
                    elif _ok:

                                float(getattr(state, "x2lite_exposure_used_today", 0.0) or 0.0) + _r0["rest"], 6)
                        PRLOG.append(dict(ev="PROBE_PROMOTE", t=now.isoformat(), sym=_r0["sym"],
                                          add_qty=_qa, done=_done,
                                          delay_min=(now - _r0["armed_at"]).total_seconds() / 60.0, **_ti))
                        PRQ["rec"] = None
